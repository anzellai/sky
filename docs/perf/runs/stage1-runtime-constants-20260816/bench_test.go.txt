func benchTree() VNode {
	vn := HtmlToVNode(buildHtmlPage(gateItems))
	assignSkyIDs(&vn, "r")
	applyStyleInjections(&vn)
	return vn
}

	b.ReportAllocs()
	b.ResetTimer()

		_ = HtmlToVNode(page)
	}
}

func BenchmarkS1_AssignSkyIDs(b *testing.B) {
	trees := make([]VNode, 0, 8)
	for i := 0; i < 8; i++ {
		trees = append(trees, HtmlToVNode(buildHtmlPage(gateItems)))
	}
	b.ReportAllocs()
	b.ResetTimer()
	for i := 0; i < b.N; i++ {
		vn := trees[i%8]
		assignSkyIDs(&vn, "r")
	}
}

func BenchmarkS1_ApplyStyleInjections(b *testing.B) {
	// The passes strip their markers, so a tree can only be measured once.
	// Pre-build a pool of already-lowered, already-id-assigned trees and
	// time ONLY the style passes.
	page := buildHtmlPage(gateItems)
	trees := make([]VNode, b.N)
	for i := range trees {
		trees[i] = HtmlToVNode(page)
		assignSkyIDs(&trees[i], "r")
	}
	b.ReportAllocs()
	b.ResetTimer()
	for i := 0; i < b.N; i++ {
		applyStyleInjections(&trees[i])
	}
}

// Same tree with the one style marker stripped: the case most real pages
// are in for at least three of the four passes.
func BenchmarkS1_ApplyStyleInjectionsNoMarkers(b *testing.B) {
	page := buildHtmlPage(gateItems)
	trees := make([]VNode, b.N)
	for i := range trees {
		trees[i] = HtmlToVNode(page)
		assignSkyIDs(&trees[i], "r")
		delete(trees[i].Children[0].Attrs, "data-sky-mq-q")
		delete(trees[i].Children[0].Attrs, "data-sky-mq-rules")
	}
	b.ReportAllocs()
	b.ResetTimer()
	for i := 0; i < b.N; i++ {
		applyStyleInjections(&trees[i])
	}
}

// BenchmarkS1_RenderBody is the render as a SESSION runs it — second and
// subsequent interactions, where the size hint is warm. The first render of
// a session has no hint and measures the same as BenchmarkS1_RenderVNode.
func BenchmarkS1_RenderBody(b *testing.B) {
	vn := benchTree()
	s := &liveSession{handlers: map[string]any{}}
	s.commitRender(&vn, renderVNode(vn, s.handlers)) // warm the hint
	b.ReportAllocs()
	b.ResetTimer()
	for i := 0; i < b.N; i++ {
		_ = s.renderBody(vn)
	}
}

// BenchmarkS1_Interaction is the whole server-side interaction below the
// Sky boundary as dispatch runs it: lower the view ADT, assign ids, run
// the four style passes, build the handler table + body string, diff.
func BenchmarkS1_Interaction(b *testing.B) {
	// `page` stands in for what the compiled Sky `view` returns. Building
	// it is ABOVE the boundary this benchmark measures (the attribution run
	// puts the user's update+view at 84% of the handler), so it is hoisted
	// out: including it would swamp the runtime cost with fixture cost.
	page := buildHtmlPage(gateItems)
	prev := benchTree()
	b.ReportAllocs()
	b.ResetTimer()
	for i := 0; i < b.N; i++ {
		next := HtmlToVNode(page)
		assignSkyIDs(&next, "r")
		applyStyleInjections(&next)
		handlers := map[string]any{}
		_ = renderVNode(next, handlers)
		_ = diffTrees(&prev, &next, nil)
	}
}

// collectIDs gathers sky-ids from the tree so the ack bench can name ids
// that really exist in it.
func collectIDs(n *VNode, out *[]string) {
	if n.Kind == "element" && n.SkyID != "" {
		*out = append(*out, n.SkyID)
	}
	for i := range n.Children {
		collectIDs(&n.Children[i], out)
	}
}

// BenchmarkS1_AckInputs — the walk that runs once any input has been typed.
// Two ids, which is what a user typing into a form actually has dirty.
// "late" places them at the end of the tree (worst case for an early exit);
// "gone" names ids that have unmounted, which forces the full walk.
func benchAck(b *testing.B, pick func(ids []string) []string) {
	vn := benchTree()
	var ids []string
	collectIDs(&vn, &ids)
	want := pick(ids)
	b.ReportAllocs()
	b.ResetTimer()
	for i := 0; i < b.N; i++ {
		b.StopTimer()
		s := &liveSession{prevTree: &vn, inputSeqs: map[string]int64{}}
		for _, id := range want {
			s.inputSeqs[id] = 1
		}
		b.StartTimer()
		_ = ackInputsForPrevTree(s)
	}
}

func BenchmarkS1_AckInputsEarly(b *testing.B) {
	benchAck(b, func(ids []string) []string { return []string{ids[2], ids[5]} })
}

func BenchmarkS1_AckInputsLate(b *testing.B) {
	benchAck(b, func(ids []string) []string {
		return []string{ids[len(ids)-2], ids[len(ids)-1]}
	})
}

func BenchmarkS1_AckInputsUnmounted(b *testing.B) {
	benchAck(b, func(ids []string) []string { return []string{"r.gone#input"} })
}
